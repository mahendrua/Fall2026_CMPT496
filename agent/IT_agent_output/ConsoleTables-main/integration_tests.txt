// Workflow: Data Ingestion and Table Construction
// Description: Standardizing the extraction and mapping of data from varied sources—including POCOs, DataTables, and Dictionaries—into a structured internal table format while enforcing schema constraints like column-row parity.
// Covered Business Rules: [1, 2, 5, 11, 14, 15, 16, 17, 18, 24, 26, 31, 33, 34, 38, 41, 42, 45, 47, 50]

using System;
using System.Collections.Generic;
using System.Data;
using System.Linq;
using Xunit;
using ConsoleTables;
using ConsoleTables_main.Tests;

[Fact]
public void VerifyDataIngestionAndTableConstructionWorkflow()
{
    // 1. POCO Collection Ingestion (Rules 1, 11, 17, 24, 42, 47)
    var pocos = new List<SamplePoco>
    {
        new SamplePoco { Id = 1, Name = "Entity A" },
        new SamplePoco { Id = 2, Name = "Entity B" }
    };
    var tableFromPoco = ConsoleTable.From(pocos);
    Assert.Equal(2, tableFromPoco.Columns.Count);
    Assert.Contains("Id", tableFromPoco.Columns);
    Assert.Contains("Name", tableFromPoco.Columns);
    Assert.Equal(2, tableFromPoco.Rows.Count);

    // 2. DataTable Ingestion with Byte-to-Base64 Conversion (Rules 11, 24, 38, 50)
    var dt = new DataTable();
    dt.Columns.Add("FileName", typeof(string));
    dt.Columns.Add("Content", typeof(byte[]));
    byte[] rawData = { 72, 101, 108, 108, 111 }; // "Hello"
    dt.Rows.Add("test.txt", rawData);
    var tableFromDt = ConsoleTable.From(dt);
    Assert.Equal("FileName", tableFromDt.Columns[0]);
    Assert.Equal("test.txt", tableFromDt.Rows[0][0]);
    // Rule 38: Check Base64 conversion
    Assert.Equal("SGVsbG8=", tableFromDt.Rows[0][1]);

    // 3. Nested Dictionary Ingestion (Rules 14, 18, 41)
    var nestedDict = new Dictionary<string, Dictionary<string, object>>
    {
        { 
            "Row_1", new Dictionary<string, object> 
            { 
                { "Col_A", "Val_A" }, 
                { "Col_B", 100 } 
            } 
        }
    };
    var tableFromDict = ConsoleTable.FromDictionary(nestedDict);
    // Rule 14/18: First column header is empty, and first cell of row is the outer key
    Assert.Equal(string.Empty, tableFromDict.Columns[0]);
    Assert.Equal("Row_1", tableFromDict.Rows[0][0]);
    Assert.Equal("Val_A", tableFromDict.Rows[0][1]);

    // 4. Manual Construction with Schema Enforcement (Rules 2, 5, 15, 16, 31, 33, 34, 45)
    // Rule 15/33: Verify exception when adding rows before columns
    var manualTable = new ConsoleTable();
    var exMissingCols = Assert.Throws<Exception>(() => manualTable.AddRow("Data"));
    Assert.Equal("Please set the columns first", exMissingCols.Message);

    // Define columns manually
    manualTable.AddColumn(new List<string> { "Index", "Status", "Symbol" });

    // Rule 16/34: Verify row count vs column count parity
    var exMismatch = Assert.Throws<Exception>(() => manualTable.AddRow(1, "Active"));
    Assert.Contains("does not match the values", exMismatch.Message);

    // Rule 5: Fluent API and Rule 45: Mixed types (Int, String, Unicode)
    manualTable.AddRow(1, "Completed", "Sparkles ✨")
               .AddRow(2, "Pending ⏳", "Unicode 哈哈哈哈")
               .Configure(options => options.NumberAlignment = Alignment.Right);

    var result = manualTable.ToString();
    Assert.Contains("Sparkles ✨", result);
    Assert.Contains("Unicode 哈哈哈哈", result);
    Assert.Contains("Count: 2", result);
}

// Workflow: Adaptive Layout and Content Wrapping
// Description: Dynamically managing table dimensions and text layout by calculating Unicode-aware widths and executing multi-line wrapping at defined word boundaries.
// Covered Business Rules: [9, 12, 21, 29, 36, 40, 46, 37]

using System;
using System.Linq;
using System.Collections.Generic;
using Xunit;
using ConsoleTables;

[Fact]
public void AdaptiveLayoutAndWrapping_ShouldHandleUnicodeAndMultiLineWrapping()
{
    // Arrange: Create a table and configure for tight constraints to force wrapping
    // Rule 40: Default MaxWidth is 40. We set it lower to test wrapping logic (Rule 21).
    var table = new ConsoleTable("Feature Description", "Status");
    table.MaxWidth = 20;
    table.SetWordBreakDelimiter(' '); // Rule 37: Designate space as word break boundary

    // Rule 36: Text content exceeding MaxWidth must wrap into new rows.
    // "Multi-line wrapping logic works!" (28 chars) should break at the space before "logic".
    // "Multi-line wrapping" is 18 chars, fitting within MaxWidth 20.
    table.AddRow("Multi-line wrapping logic works!", "Verified");

    // Rule 9, 29, 46: Add content with Unicode and special characters that challenge width calculation.
    // "Accented René ✨" contains accented 'é' and emoji '✨'.
    table.AddRow("Accented René ✨", "Valid");

    // Act: Generating output triggers the internal formatting and wrapping logic (SetFormats)
    string output = table.ToString();

    // Assert: Check row insertion via wrapping (Rule 12, 36)
    // The row with "Multi-line wrapping logic works!" should split into two rows.
    // 1st row: "Multi-line wrapping"
    // 2nd row: "logic works!"
    Assert.True(table.Rows.Count >= 3, $"Expected at least 3 rows due to wrapping, but found {table.Rows.Count}");
    
    var wrappedRowPart1 = table.Rows[0][0].ToString();
    var wrappedRowPart2 = table.Rows[1][0].ToString();

    Assert.Equal("Multi-line wrapping", wrappedRowPart1);
    Assert.Equal("logic works!", wrappedRowPart2);

    // Assert: Rule 9, 29, 46 - Verify visual integrity and Unicode-aware alignment
    // Ensure the output contains the Unicode data and is bounded by delimiters '|'
    Assert.Contains("| Accented René ✨   | Valid    |", output);

    // Verify visual alignment structure (every line with content should have aligned borders)
    var lines = output.Split(new[] { Environment.NewLine }, StringSplitOptions.RemoveEmptyEntries);
    foreach (var line in lines)
    {
        if (line.Contains("|"))
        {
            var trimmedLine = line.Trim();
            Assert.StartsWith("|", trimmedLine);
            Assert.EndsWith("|", trimmedLine);
            
            // Rule 9: Column alignment check - check for fixed length if logic is working correctly
            // All content lines should have the same character count for box integrity.
            // We compare against the header line length.
            var headerLine = lines.First(l => l.Contains("Feature Description")).Trim();
            Assert.Equal(headerLine.Length, trimmedLine.Length);
        }
    }

    // Rule 40 verification: MaxWidth is a public configurable property
    Assert.Equal(20, table.MaxWidth);
}

// Workflow: Data Alignment and Visual Styling
// Description: Configuring and applying uniform formatting policies, specifically numeric right-alignment and cell padding, to improve the readability of statistical data across all ingestion methods.
// Covered Business Rules: [4, 7, 23, 25, 30, 35, 43, 48]

using System;
using System.Collections.Generic;
using System.Linq;
using Xunit;
using ConsoleTables;

[Fact]
public void DataAlignmentAndVisualStyling_ShouldApplyUniformFormattingAcrossAllIngestionMethodsAndOutputStyles()
{
    // Arrange: Create test data containing numeric and non-numeric types
    var testData = new[]
    {
        new { Name = "Alexandre", Age = 36 },
        new { Name = "Otto", Age = 52 }
    };

    // Ingestion Method 1: Create table from an object collection (POCO/Anonymous Objects)
    var tableFromPoco = ConsoleTable.From(testData);

    // Ingestion Method 2: Create table and add rows manually
    var tableManual = new ConsoleTable("Name", "Age");
    foreach (var item in testData)
    {
        tableManual.AddRow(item.Name, item.Age);
    }

    // Act: Configure the formatting policy (Numeric Right-Alignment)
    // This configuration demonstrates configurable alignment and treats numeric data differently from text (Rules 4, 23, 30, 43, 48)
    Action<ConsoleTableOptions> configuration = options =>
    {
        options.NumberAlignment = Alignment.Right;
        options.EnableCount = true;
    };

    tableFromPoco.Configure(configuration);
    tableManual.Configure(configuration);

    // Retrieve different visual styles as strings (Rules 7, 25)
    var defaultStyle = tableFromPoco.ToString();
    var manualDefaultStyle = tableManual.ToString();
    var markdownStyle = tableFromPoco.ToMarkDownString();
    var minimalStyle = tableFromPoco.ToMinimalString();

    // Assert: Verify visual presentation is applied consistently regardless of ingestion method (Rule 7, 25)
    // The Age column width is 3 (header "Age" is 3, value "36" is 2). 
    // With Right Alignment, "36" in width 3 is " 36". 
    // The ConsoleTable formatting template adding a leading space results in "  36 ".
    Assert.Equal(defaultStyle, manualDefaultStyle);
    Assert.Contains("| Name      | Age |", defaultStyle);
    Assert.Contains("| Alexandre |  36 |", defaultStyle);
    Assert.Contains("| Otto      |  52 |", defaultStyle);

    // Assert: Verify consistency across output styles (Rule 25, 35)
    // Markdown Output
    Assert.Contains("| Name      | Age |", markdownStyle);
    Assert.Contains("|-----------|-----", markdownStyle);
    Assert.Contains("| Alexandre |  36 |", markdownStyle);

    // Minimal Output (Rule 25, 30)
    // Minimal style typically uses space-based separation without pipes
    Assert.Contains("Name       Age", minimalStyle);
    Assert.Contains("Alexandre  36", minimalStyle);
    
    // Verify the configuration was preserved in the Options object
    Assert.Equal(Alignment.Right, tableFromPoco.Options.NumberAlignment);
    Assert.True(tableFromPoco.Options.EnableCount);
}

// Workflow: Metadata Summarization
// Description: Calculating and displaying optional summary metadata, such as total record counts, as a standardized footer in the generated reports.
// Covered Business Rules: [19, 32, 39, 44, 52]

using System;
using System.IO;
using System.Linq;
using Xunit;
using ConsoleTables;

[Fact]
public void Workflow_MetadataSummarization_VerifyDefaultAndConfigurableCountFooter()
{
    // Step 1: Arrange - Create a table and add rows to ensure there is data to count
    var table = new ConsoleTable("Feature", "Status");
    table.AddRow("Metadata Summarization", "Enabled");
    table.AddRow("Row Count Footer", "Default");
    table.AddRow("Configuration Toggle", "Tested");

    // Step 2: Act - Get default string representation (Rule 19, 39, 52)
    // The default behavior should include the row count footer because EnableCount is true by default
    var outputWithCount = table.ToString();

    // Step 3: Assert - Verify the count footer is present and accurate
    Assert.Contains("Count: 3", outputWithCount);

    // Step 4: Act - Toggle the summary metadata off via configuration (Rule 32, 44)
    table.Configure(options => options.EnableCount = false);
    var outputWithoutCount = table.ToString();

    // Step 5: Assert - Verify the count footer is no longer present
    Assert.DoesNotContain("Count: 3", outputWithoutCount);
    Assert.DoesNotContain("Count:", outputWithoutCount);

    // Step 6: Act - Verify consistency across visual styles (Rule 19)
    // Re-enable and verify it persists in the configuration state
    table.Configure(o => 
    {
        o.EnableCount = true;
        o.NumberAlignment = Alignment.Right;
    });

    var markdownOutput = table.ToMarkDownString();
    var alternativeOutput = table.ToStringAlternative();

    // Assert state preservation in options object
    Assert.True(table.Options.EnableCount);
    Assert.Equal(Alignment.Right, table.Options.NumberAlignment);

    // Standard string representation should still contain the count
    Assert.Contains("Count: 3", table.ToString());
}
